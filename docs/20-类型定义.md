# 类型定义

TypeScript 类型需要显式 import。统一从 `@flecblog/core-nuxt` 导入。

```ts
import type { Article, Comment, UserInfo } from '@flecblog/core-nuxt'
```

## 文章

```ts
interface Article {
  id: number
  title: string
  slug: string
  content: string
  summary: string
  ai_summary?: string
  cover: string
  location: string
  is_top: boolean
  is_essence: boolean
  is_outdated: boolean
  view_count: number
  comment_count: number
  url: string
  publish_time: string
  update_time: string
  created_at: string
  category?: Category
  tags?: Tag[]
  prev?: ArticleNav
  next?: ArticleNav
}

interface ArticleNav {
  title: string
  url: string
}

interface ArticleQuery {
  page?: number
  page_size?: number
  category?: string
  tag?: string
  year?: string
  month?: string
}
```

## 评论

```ts
type CommentTargetType = 'article' | 'page'

interface Comment {
  id: number
  content: string
  is_deleted: boolean
  is_pinned: boolean
  parent_id: number | null
  created_at: string
  location?: string
  browser?: string
  os?: string
  user: {
    role: UserRole
    badge?: string
    id: number
    email_hash: string
    nickname: string
    avatar: string
    website?: string
  }
  reply_user?: {
    role: UserRole
    badge?: string
    id: number
    email_hash: string
    nickname: string
    avatar: string
    website?: string
  }
  replies: Comment[]
}

interface CreateCommentParams {
  target_type: CommentTargetType
  target_key: string | number
  content: string
  parent_id?: number
  nickname?: string  // 游客昵称（未登录时）
  email?: string     // 游客邮箱（未登录时）
  website?: string   // 游客网站（未登录时）
}

interface GuestInfo {
  nickname?: string
  email?: string
  website?: string
}

interface GetCommentsParams {
  target_type: CommentTargetType
  target_key: string | number
  page?: number
  page_size?: number
}

interface FlatComment {
  comment: Comment
  depth: number
}

interface EmojiItem {
  key: string
  val: string
}

interface EmojiGroup {
  name: string
  type: 'emoji' | 'image' | 'emoticon'
  items: EmojiItem[]
}
```
